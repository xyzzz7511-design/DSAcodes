#include <iostream>
using namespace std;

struct Student
{
    int id;
    char name[50];
    float cgpa;
};

int main()
{
    Student *s = new Student[1];
    int n = 0, choice, key;

    do
    {
        cout << "\n--- STUDENT DATABASE ---";
        cout << "\n1. Add Student";
        cout << "\n2. Display Students";
        cout << "\n3. Linear Search";
        cout << "\n4. Binary Search";
        cout << "\n5. Exit";
        cout << "\nEnter choice: ";
        cin >> choice;

        switch(choice)
        {
            case 1:
            {
                Student *temp = new Student[n + 1];

                for(int i = 0; i < n; i++)
                    temp[i] = s[i];

                delete[] s;
                s = temp;

                cout << "Enter ID: ";
                cin >> s[n].id;

                cout << "Enter Name: ";
                cin >> s[n].name;

                cout << "Enter CGPA: ";
                cin >> s[n].cgpa;

                n++;
                cout << "Student Added Successfully!";
                break;
            }

            case 2:
                for(int i = 0; i < n; i++)
                {
                    cout << "\nID: " << s[i].id;
                    cout << "\nName: " << s[i].name;
                    cout << "\nCGPA: " << s[i].cgpa;
                    cout << endl;
                }
                break;

            case 3:
            {
                cout << "Enter ID to search: ";
                cin >> key;

                int found = 0;

                for(int i = 0; i < n; i++)
                {
                    if(s[i].id == key)
                    {
                        cout << "Student Found!";
                        cout << "\nID: " << s[i].id;
                        cout << "\nName: " << s[i].name;
                        cout << "\nCGPA: " << s[i].cgpa;
                        found = 1;
                        break;
                    }
                }

                if(found == 0)
                    cout << "Student Not Found!";

                break;
            }

            case 4:
            {
                // Sort records by ID
                for(int i = 0; i < n-1; i++)
                {
                    for(int j = 0; j < n-i-1; j++)
                    {
                        if(s[j].id > s[j+1].id)
                        {
                            Student temp = s[j];
                            s[j] = s[j+1];
                            s[j+1] = temp;
                        }
                    }
                }

                cout << "Enter ID to search: ";
                cin >> key;

                int low = 0, high = n-1;
                int found = 0;

                while(low <= high)
                {
                    int mid = (low + high) / 2;

                    if(s[mid].id == key)
                    {
                        cout << "Student Found!";
                        cout << "\nID: " << s[mid].id;
                        cout << "\nName: " << s[mid].name;
                        cout << "\nCGPA: " << s[mid].cgpa;
                        found = 1;
                        break;
                    }
                    else if(s[mid].id < key)
                        low = mid + 1;
                    else
                        high = mid - 1;
                }

                if(found == 0)
                    cout << "Student Not Found!";

                break;
            }

            case 5:
                cout << "Program Exited!";
                break;

            default:
                cout << "Invalid Choice!";
        }

    } while(choice != 5);

    delete[] s;

    return 0;
}        


1. Title
Design a C++ program to maintain a Student Database that performs the following operations: (1) Add and
store student details (ID, Name, CGPA) using dynamically allocated memory. (2) Expand the student list
using realloc() as new entries are added. (3) Implement Linear Search and Binary Search to find student
records by ID.
2. Theory
Student Database: A Student Database is used to store and manage student information such as Student
ID, Name and CGPA.
Dynamic Memory Allocation: Dynamic Memory Allocation is a technique used to allocate memory during
program execution as required.
realloc(): The realloc() function is used to increase or decrease the size of previously allocated memory
when new student records are added.
Linear Search: Linear Search is a searching technique that checks each element one by one until the
required element is found or the list ends. Time Complexity: O(n).
Binary Search: Binary Search is an efficient searching technique that works on a sorted list. It repeatedly
divides the list into two halves to find the required element. Time Complexity: O(log n).
3. Algorithms
Algorithm 1: Linear Search
1. Start.
2. Accept the Student ID to be searched.
3. Initialize the index to 0.
4. Compare the search ID with each student's ID.
5. If the ID matches, display the student details.
6. Otherwise, move to the next record.
7. If the ID is not found after checking all records, display “Student Not Found”.
8. Stop.
Algorithm 2: Binary Search
1. Start.
2. Sort student records in ascending order of Student ID.
3. Accept the Student ID to be searched.
4. Initialize low = 0 and high = n - 1.
5. Calculate the middle position using mid = (low + high) / 2.
6. Compare the middle student's ID with the search ID.
7. If both IDs are equal, display the student details.
8. If the search ID is smaller, set high = mid - 1.
9. If the search ID is greater, set low = mid + 1.
10. Repeat until the ID is found or low > high. Page 2
11. If the ID is not found, display “Student Not Found”.
12. Stop.
