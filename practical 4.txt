#include <iostream>
using namespace std;

#define SIZE 5

int queue[SIZE];
int front = -1, rear = -1;

// Insertion
void insert(int value)
{
    if ((front == 0 && rear == SIZE - 1) ||
        (rear + 1 == front))
    {
        cout << "Queue is Full!" << endl;
        return;
    }

    if (front == -1)
        front = 0;

    rear = (rear + 1) % SIZE;
    queue[rear] = value;

    cout << "Inserted: " << value << endl;
}

// Deletion
void Delete()
{
    if (front == -1)
    {
        cout << "Queue is Empty!" << endl;
        return;
    }

    cout << "Deleted: " << queue[front] << endl;

    if (front == rear)
    {
        front = rear = -1;
    }
    else
    {
        front = (front + 1) % SIZE;
    }
}

// Display
void display()
{
    if (front == -1)
    {
        cout << "Queue is Empty!" << endl;
        return;
    }

    cout << "Queue Elements: ";

    int i = front;

    while (true)
    {
        cout << queue[i] << " ";

        if (i == rear)
            break;

        i = (i + 1) % SIZE;
    }

    cout << endl;
}

// Main Function
int main()
{
    int choice, value;

    while (true)
    {
        cout << "\n--- Circular Queue Menu ---" << endl;
        cout << "1. Insertion" << endl;
        cout << "2. Deletion" << endl;
        cout << "3. Display" << endl;
        cout << "Enter your choice: ";
        cin >> choice;

        switch (choice)
        {
            case 1:
                cout << "Enter value: ";
                cin >> value;
                insert(value);
                break;

            case 2:
                Delete();
                break;

            case 3:
                display();
                break;

            default:
                cout << "Invalid Choice!" << endl;
        }
    }

    return 0;
}

Theory:::::
A circular queue is a linear data structure that follows the FIFO (First In, First Out) principle. It connects the last position of the array back to the first position, forming a circle.
It efficiently utilizes memory by reusing empty spaces created after deletion.
Main operations:
Enqueue: Inserts an element into the queue.
Dequeue: Removes an element from the queue.
Display: Displays all elements of the queue.

Algorithm:::::::

Enqueue (Insertion):
Start.
Check whether the queue is full.
If full, display overflow.
Otherwise, update the rear position using circular increment.
Insert the new element.
Stop.

Dequeue (Deletion):
Start.
Check whether the queue is empty.
If empty, display underflow.
Otherwise, remove the front element.
Update the front position.
Stop.

Display:
Check whether the queue is empty.
If empty, display a message.
Otherwise, traverse from front to rear using circular increment.
Display all elements.
Stop.

Conclusion:::::

Thus, a circular queue was successfully implemented using an array, and insertion, deletion, and display operations were performed.
